    // ⚠️ CAMBIAR ESTE NÚMERO SEGÚN EL TEMA (1, 2, 14, 17, etc.)

    // Guardar nota e intentos en Firebase
    window.saveTestResults = async function(scoreNum) {
      if (window.testSavedForThisSession) return;
      window.testSavedForThisSession = true;

      try {
        const topicRef = ref(db, `progress/${CURRENT_TOPIC_ID}`);
        const snapshot = await get(topicRef);
        const data = snapshot.val() || {};

        const newAttempts = (data.attempts || 0) + 1;
        const currentBest = data.bestScore || 0;
        const newBest = Math.max(currentBest, parseFloat(scoreNum.toFixed(1)));

        await set(topicRef, {
          attempts: newAttempts,
          bestScore: newBest
        });
      } catch (e) {
        console.error("Error guardando en Firebase:", e);
      }
    };

    // Borrar historial de este tema en la nube
    window.resetTopicStats = async function() {
      if (confirm(`¿Quieres borrar los intentos y la nota máxima del Tema ${CURRENT_TOPIC_ID} en la nube?`)) {
        try {
          const topicRef = ref(db, `progress/${CURRENT_TOPIC_ID}`);
          await remove(topicRef);
          alert('Progreso borrado en la nube.');
          location.reload();
        } catch (e) {
          alert('Error al intentar borrar el progreso.');
        }
      }
    };
  </script>

    function updateStats() {
      document.getElementById('progress').textContent = `📝 ${answeredCount}/${activeData.length}`;
      
      const accuracyPct = answeredCount > 0 ? Math.round((correctCount / answeredCount) * 100) : 0;
      document.getElementById('accuracy').textContent = `🎯 ${accuracyPct}%`;

      const numericScore = activeData.length > 0 ? (correctCount / activeData.length) * 10 : 0;
      document.getElementById('score').textContent = `⭐ ${numericScore.toFixed(2)}/10`;

      // Guardar en Firebase automáticamente al responder la última pregunta
      if (answeredCount > 0 && answeredCount === activeData.length && typeof window.saveTestResults === 'function') {
        window.saveTestResults(numericScore);
      }
    }
